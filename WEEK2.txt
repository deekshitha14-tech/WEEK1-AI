import heapq
def water_jug(A, B, C):
    pq = [(0, 0, 0, 0)]          # (f, g, jug1, jug2)
    parent = {(0, 0): None}
    visited = set()
    while pq:
        _, g, x, y = heapq.heappop(pq)
        if (x, y) in visited:
            continue
        visited.add((x, y))
        # Goal check
        if x == C or y == C:
            path = []
            while True:
                path.append((x, y))
                if parent[(x, y)] is None:
                    break
                x, y = parent[(x, y)]
            print("Path:", path[::-1])
            return
        # All possible next states
        next_states = [
            (A, y),                      # fill jug1
            (x, B),                      # fill jug2
            (0, y),                      # empty jug1
            (x, 0),                      # empty jug2
            (x - min(x, B - y), y + min(x, B - y)),  # jug1 → jug2
            (x + min(y, A - x), y - min(y, A - x))   # jug2 → jug1
        ]
        for nx, ny in next_states:
            if (nx, ny) not in visited:
                parent[(nx, ny)] = (x, y)
                h = min(abs(nx - C), abs(ny - C))
                heapq.heappush(pq, (g + 1 + h, g + 1, nx, ny))
    print("Not Possible")

import heapq
def fuel_blend_astar_diff_strategy(A,B,target):
    # User input
    pq = [(0 + 0, 0, (0,0))]  # (f=g+h, g, (A_amount,B_amount))
    parent = {(0,0): None}
    visited = set()
    while pq:
        _, g, (x, y) = heapq.heappop(pq)
        if (x, y) in visited: continue
        visited.add((x, y))
        # Goal
        if x == target:
            path = []
            while (x, y) in parent:
                path.append((x, y))
                if parent[(x, y)] is None: break
                x, y = parent[(x, y)]
            print("Path:", path[::-1])
            return
        # Possible moves: fill, empty, pour
        moves = [
            (A, y),                   # fill A
            (x, B),                   # fill B
            (0, y),                   # empty A
            (x, 0),                   # empty B
            (x - min(x, B - y), y + min(x, B - y)),  # A -> B
            (x + min(y, A - x), y - min(y, A - x))   # B -> A
        ]
        for nx, ny in moves:
            if (nx, ny) not in visited:
                parent[(nx, ny)] = (x, y)
                h = abs(nx - target) + abs(ny)  # <-- Different heuristic
                heapq.heappush(pq, (g + 1 + h, g + 1, (nx, ny)))
    print("Not Possible")
